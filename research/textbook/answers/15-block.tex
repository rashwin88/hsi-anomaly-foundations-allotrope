\begin{solution}{ex:block:1}
\emph{Mean.} $(2.0 + 4.0 + 4.0 + 6.0)/4 = 16.0/4 = 4.0$.

\emph{Deviations and squares.}
\begin{align*}
  2.0 - 4.0 &= -2.0, & (-2.0)^2 &= 4.0\\
  4.0 - 4.0 &= \phantom{-}0.0, & 0.0^2 &= 0.0\\
  4.0 - 4.0 &= \phantom{-}0.0, & 0.0^2 &= 0.0\\
  6.0 - 4.0 &= \phantom{-}2.0, & 2.0^2 &= 4.0
\end{align*}
\emph{Variance.} $(4.0 + 0.0 + 0.0 + 4.0)/4 = 8.0/4 = 2.0$. \emph{Standard deviation.}
$\sqrt{2.0} = 1.4142$.

\emph{Outputs.} $-2.0/1.4142 = -1.4142$, $0/1.4142 = 0$, $0$, $2.0/1.4142 = 1.4142$. So
$\LN = (-1.4142, 0, 0, 1.4142)$. Check: sum 0; squares $2 + 0 + 0 + 2 = 4$, average 1.

$(10, 20, 20, 30) = 5\times(2, 4, 4, 6)$, a positive rescaling, so by the invariance of
\cref{sec:block:ln} its LayerNorm is the same $(-1.4142, 0, 0, 1.4142)$.
\end{solution}

\begin{solution}{ex:block:2}
$\GELU(2) = 2\times\Phi(2) = 2\times0.9772 = 1.9545$ (to four places from $\Phi(2) = 0.97725$).
By symmetry $\Phi(-2) = 1 - 0.9772 = 0.0228$, so $\GELU(-2) = -2\times0.0228 = -0.0455$.

The fraction passed is $\Phi(x)$ itself: 97.7\% at $x = 2$ and 2.3\% at $x = -2$. ReLU would pass
100\% and 0\%.
\end{solution}

\begin{solution}{ex:block:3}
\emph{Without residuals}: $(-0.1)^{16} = 0.1^{16} = 10^{-16}$ (an even power, so positive). The
signal reaching the first layer is one part in ten million billion: nothing.

\emph{With residuals}: each factor is $1 + f' = 1 - 0.1 = 0.9$, and $0.9^{16}$:
\begin{align*}
  0.9^2 &= 0.81, & 0.9^4 &= 0.81^2 = 0.6561, & 0.9^8 &= 0.6561^2 = 0.4305, & 0.9^{16} &= 0.4305^2 = 0.1853 .
\end{align*}
About a fifth of the signal arrives. Only the residual stack lets the first layer learn.
\end{solution}

\begin{solution}{ex:block:4}
$C = 64$, hidden width $4\times64 = 256$.
\begin{align*}
  \text{fc1: } 64\times256 + 256 &= 16{,}384 + 256 = 16{,}640\\
  \text{dwconv: } 256\times9 + 256 &= 2{,}304 + 256 = 2{,}560\\
  \text{fc2: } 256\times64 + 64 &= 16{,}384 + 64 = 16{,}448\\
  \text{total: } 16{,}640 + 2{,}560 + 16{,}448 &= 35{,}648 .
\end{align*}
Check: $8\times64^2 + 45\times64 = 8\times4{,}096 + 2{,}880 = 32{,}768 + 2{,}880 = 35{,}648$.
\end{solution}

\begin{solution}{ex:block:5}
$C = 96$, hidden width $4\times96 = 384$.
\begin{align*}
  \text{norm1: } 2\times96 &= 192\\
  \text{q, k, v, proj: } 4\times(96\times96 + 96) = 4\times(9{,}216 + 96) = 4\times9{,}312 &= 37{,}248\\
  \text{reduction conv: } 96\times96\times64 + 96 = 589{,}824 + 96 &= 589{,}920\\
  \text{reduction norm: } 2\times96 &= 192\\
  \text{norm2: } 2\times96 &= 192\\
  \text{fc1: } 96\times384 + 384 = 36{,}864 + 384 &= 37{,}248\\
  \text{dwconv: } 384\times9 + 384 = 3{,}456 + 384 &= 3{,}840\\
  \text{fc2: } 384\times96 + 96 = 36{,}864 + 96 &= 36{,}960
\end{align*}
Sum: $192 + 37{,}248 = 37{,}440$; $+ 589{,}920 = 627{,}360$; $+ 192 = 627{,}552$; $+ 192 = 627{,}744$;
$+ 37{,}248 = 664{,}992$; $+ 3{,}840 = 668{,}832$; $+ 36{,}960 = 705{,}792$. Confirmed.

Ratio: $705{,}792/79{,}616 = 8.86$. Every weight matrix is $C\times C$ (or $C\times C\times k\times k$),
so tripling $C$ multiplies it by $3^2 = 9$; but biases, LayerNorms and the depthwise weights are
proportional to $C$ and grow only threefold, which pulls the ratio a little under 9.
\end{solution}

\begin{solution}{ex:block:6}
With $x = 1$ and $1/(1-p) = 1/0.6 = 1.6667$, the dropped value is $1.6667$ with probability 0.6 and
0 with probability 0.4.
\begin{align*}
  \E &= 0.6\times1.6667 + 0.4\times0 = 1.0000\\
  \E[\cdot^2] &= 0.6\times1.6667^2 + 0.4\times0 = 0.6\times2.7778 = 1.6667\\
  \operatorname{Var} &= 1.6667 - 1.0000^2 = 0.6667 .
\end{align*}
In general the variance is $x^2\,p/(1-p)$: the mean is preserved but dropout adds noise, which is
what forces the network not to rely on any single value.
\end{solution}
