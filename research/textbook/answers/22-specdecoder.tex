\begin{solution}{ex:specdecoder:1}
With $d = 16$, $D = 32$:
\begin{align*}
  \text{query from wavelength, Linear(16, 16)}:&\quad 16\times16 + 16 = 272\\
  \text{channel value stretch, Linear(1, 16)}:&\quad 16 + 16 = 32\\
  \text{channel identities, table (32, 16)}:&\quad 32\times16 = 512\\
  \text{Q, K, V, O}:&\quad 4\times272 = 1{,}088\\
  \text{to reflectance, Linear(16, 1)}:&\quad 16 + 1 = 17
\end{align*}
Total: $272 + 32 + 512 + 1{,}088 + 17 = 304 + 512 + 1{,}088 + 17 = 816 + 1{,}088 + 17 = 1{,}904 + 17 = 1{,}921$.
Formula $5d^2 + Dd + 8d + 1$: $5\times256 + 32\times16 + 8\times16 + 1 = 1{,}280 + 512 + 128 + 1 = 1{,}921$.
\end{solution}

\begin{solution}{ex:specdecoder:2}
\begin{align*}
  \sigma(1) &= 1/(1 + e^{-1}) = 1/(1 + 0.3679) = 1/1.3679 = 0.7311,\\
  \sigma(-1) &= 1/(1 + e^{1}) = 1/(1 + 2.7183) = 1/3.7183 = 0.2689 .
\end{align*}
They sum to 1, as $\sigma(z) + \sigma(-z)$ always does.

Gates $[0.88, 0.88, 0.12, 0.12]$:
\begin{align*}
  \vect{x}' &= [0.88\times1.2,\ 0.88\times1.0,\ 0.12\times(-0.4),\ 0.12\times(-0.6)]\\
  &= [1.056,\ 0.880,\ -0.048,\ -0.072],\\
  y_0 &= 0.5\times1.056 + 0.5\times0.880 = 0.528 + 0.440 = 0.968,\\
  y_1 &= 0.5\times(-0.048) + 0.5\times(-0.072) = -0.024 - 0.036 = -0.060 .
\end{align*}
Now the second output barely listens to bands 2 and 3.
\end{solution}

\begin{solution}{ex:specdecoder:3}
\begin{align*}
  \text{first layer}, 165\to32:&\quad 165\times32 + 32 = 5{,}280 + 32 = 5{,}312\\
  \text{second layer}, 32\to165:&\quad 32\times165 + 165 = 5{,}280 + 165 = 5{,}445\\
  \text{total}:&\quad 5{,}312 + 5{,}445 = 10{,}757 .
\end{align*}
The real module reports 10,757. Leak:
\begin{align*}
  \sigma(-4) &= 1/(1 + e^{4}) = 1/(1 + 54.598) = 1/55.598 = 0.0180,\\
  0.0180\times50 &= 0.90 .
\end{align*}
The cross-attention compressor gives a masked band weight exactly 0, so nothing leaks.
\end{solution}

\begin{solution}{ex:specdecoder:4}
\begin{align*}
  (2500 - 400)/50 &= 2100/50 = 42, & \lceil 42\rceil &= 42 \text{ bins}.
\end{align*}
Here 2100 is a whole multiple of 50, so the last bin is full width. For 864\,nm:
\begin{align*}
  (864 - 400)/50 &= 464/50 = 9.28, & \lfloor 9.28\rfloor &= 9 .
\end{align*}
Bin 9 covers $400 + 50\times9 = 850$ to $400 + 50\times10 = 900$\,nm.
\end{solution}

\begin{solution}{ex:specdecoder:5}
Spectrum $[0.3, 0.5]$:
\begin{align*}
  \tfrac12\bigl(0.3[1, 0] + 0.5[1, 1]\bigr) &= \tfrac12\bigl([0.3, 0] + [0.5, 0.5]\bigr) = \tfrac12[0.8, 0.5] = [0.4, 0.25].
\end{align*}
Spectrum $[0.5, 0.3]$:
\begin{align*}
  \tfrac12\bigl(0.5[1, 0] + 0.3[1, 1]\bigr) &= \tfrac12\bigl([0.5, 0] + [0.3, 0.3]\bigr) = \tfrac12[0.8, 0.3] = [0.4, 0.15].
\end{align*}
Different, so the order of the values survives. The first component is the sum (both 0.8, halved to 0.4), which is all a plain average keeps; the second component, carried only by band 2, tells the two apart. With a shared weight both spectra would have given the same token.
\end{solution}

\begin{solution}{ex:specdecoder:6}
Band 0 is dead, so its value is multiplied by validity 0:
\begin{align*}
  \text{vals} &= [99\times0,\ 1.0,\ -0.4,\ -0.6] = [0,\ 1.0,\ -0.4,\ -0.6],\\
  \text{sum} &= 1.0[0.5, 0] + (-0.4)[0, 0.5] + (-0.6)[0, 0.5]\\
  &= [0.5, 0] + [0, -0.2] + [0, -0.3] = [0.5, -0.5],\\
  \text{mean} &= [0.5, -0.5]/3 = [0.1667, -0.1667] .
\end{align*}
Three valid bands, so divide by 3.
\end{solution}

\begin{solution}{ex:specdecoder:7}
\begin{align*}
  (2.8 - 0.4)/(32 - 4) &= 2.4/28 = 0.086 \text{ GiB per extra patch},\\
  0.39/0.086 &= 4.5 .
\end{align*}

\end{solution}
