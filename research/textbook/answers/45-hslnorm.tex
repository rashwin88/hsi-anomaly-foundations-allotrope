\begin{solution}{ex:hslnorm:1}
Band 1 holds 0.15, 0.20, 0.25: $m_1=0.15$, $M_1-m_1=0.10$. Band 2 holds 0.25, 0.50, 0.55: $m_2=0.25$, $M_2-m_2=0.30$.
\begin{align*}
x'_1 &= (0.20-0.15)/0.10 = 0.05/0.10 = 0.5,\\
x'_2 &= (0.50-0.25)/0.30 = 0.25/0.30 = 0.8333 .
\end{align*}
A third input for the same spectrum: $(0.5,1.0)$ in $S_1$, $(0.4286,0.75)$ in $S_2$, $(0.5,0.8333)$ in $S_3$.
\end{solution}

\begin{solution}{ex:hslnorm:3}
\begin{align*}
\vect{u}' &= (0.20/0.20,\ 0.50/0.10)=(1,\ 5),\\
\vect{v}' &= (0.30/0.20,\ 0.40/0.10)=(1.5,\ 4),\\
\vect{u}'\cdot\vect{v}' &= 1\times1.5+5\times4 = 1.5+20 = 21.5,\\
\norm{\vect{u}'} &= \sqrt{1+25} = \sqrt{26} = 5.0990, \qquad \norm{\vect{v}'} = \sqrt{2.25+16} = \sqrt{18.25} = 4.2720,\\
\cos\theta' &= 21.5/(5.0990\times4.2720) = 21.5/21.783 = 0.9870, \qquad \theta' = 9.25^\circ .
\end{align*}
\end{solution}
