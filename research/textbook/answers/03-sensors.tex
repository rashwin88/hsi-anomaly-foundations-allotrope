\begin{solution}{ex:sensors:1}
Step $10$:
\begin{align*}
\frac{2{,}500 - 400}{10} + 1 &= \frac{2{,}100}{10} + 1 = 210 + 1 = 211 \text{ rungs.}
\end{align*}
Step $5$:
\begin{align*}
\frac{2{,}500 - 400}{5} + 1 &= \frac{2{,}100}{5} + 1 = 420 + 1 = 421 \text{ rungs.}
\end{align*}
Halving the step does not exactly double the count, because of the fencepost $+1$:
$2 \times 211 = 422 \ne 421$.
\end{solution}

\begin{solution}{ex:sensors:2}
By \cref{eq:sensors:count}:
\begin{align*}
\lceil 1{,}125/10 \rceil &= \lceil 112.5 \rceil = 113,\\
\lfloor 1{,}160/10 \rfloor &= \lfloor 116 \rfloor = 116,\\
116 - 113 + 1 &= 4 \quad (1{,}130,\ 1{,}140,\ 1{,}150,\ 1{,}160).
\end{align*}
The windows now remove $0 + 6 + 4 + 11 + 16 = 37$ rungs, so the grid would have
$200 - 37 = 163$ bands. Every saved model, statistics file and shard built for $165$ would then be
misaligned, which is the wire-format warning of \cref{sec:sensors:wire}.
\end{solution}

\begin{solution}{ex:sensors:3}
Band $70$: $62 \le 70 < 81$, so $(c_0, \lambda_0) = (62, 1{,}160)$:
\[ \lambda(70) = 1{,}160 + 10 \times (70 - 62) = 1{,}160 + 80 = 1{,}240 \text{ nm}. \]
Band $164$: $115 \le 164$, so $(c_0, \lambda_0) = (115, 1{,}960)$:
\[ \lambda(164) = 1{,}960 + 10 \times 49 = 1{,}960 + 490 = 2{,}450 \text{ nm}, \]
the last band, as expected. Index of $2{,}210$ nm (fifth segment):
\[ c = 115 + (2{,}210 - 1{,}960)/10 = 115 + 25 = 140. \]
\end{solution}

\begin{solution}{ex:sensors:4}
With $(c, i, j) = (2, 1, 3)$, $C = 4$, $H = 3$, $W = 5$:
\begin{align*}
\text{BSQ: } & c\,HW + i\,W + j = 2 \times 15 + 1 \times 5 + 3 = 30 + 5 + 3 = 38,\\
\text{BIL: } & i\,CW + c\,W + j = 1 \times 20 + 2 \times 5 + 3 = 20 + 10 + 3 = 33,\\
\text{BIP: } & i\,WC + j\,C + c = 1 \times 20 + 3 \times 4 + 2 = 20 + 12 + 2 = 34.
\end{align*}
\end{solution}

\begin{solution}{ex:sensors:5}
\begin{align*}
21{,}840 - 16{,}384 &= 5{,}456 \quad (2^{14}),\\
5{,}456 - 4{,}096 &= 1{,}360 \quad (2^{12}),\\
1{,}360 - 1{,}024 &= 336 \quad (2^{10}),\\
336 - 256 &= 80 \quad (2^{8}),\\
80 - 64 &= 16 \quad (2^{6}),\\
16 - 16 &= 0 \quad (2^{4}).
\end{align*}
Bit $4$ (cloud shadow) is set, so the builder marks the pixel \emph{invalid}. (It is
$21{,}824$ from \cref{we:sensors:qa} plus $2^4 = 16$.)

EnMAP: $\rho = 1{,}830 \times 0.0001 = 0.183$, an $18.3\%$ reflectance.
\end{solution}
